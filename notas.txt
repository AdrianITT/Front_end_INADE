from rest_framework.decorators import api_view, parser_classes, permission_classes, throttle_classes
from rest_framework.permissions import AllowAny
from rest_framework.parsers import MultiPartParser, FormParser
from rest_framework.response import Response
from rest_framework import status
from rest_framework.throttling import AnonRateThrottle
from django.db import transaction
from django.core.exceptions import ValidationError
from django.conf import settings
from ...models import (
    Organizacion, User, InfoSistema, InfoOrdenTrabajo,
    InfoCotizacion, ImagenMarcaAgua, Iva, TipoMoneda
)
import os, json, imghdr, bleach, logging

# ============================================================
# ⚙️ CONFIGURACIÓN GENERAL DE SEGURIDAD
# ============================================================

logger = logging.getLogger("registro_wizard")

# 🔒 Límite de peticiones por IP (anti-bot / anti-flood)
class RegistroWizardThrottle(AnonRateThrottle):
    rate = "3/hour"  # máximo 3 intentos por IP por hora


# ============================================================
# 🧩 Funciones auxiliares
# ============================================================

def limpiar_texto(texto):
    """Evita inyección HTML o scripts."""
    if not texto:
        return ""
    return bleach.clean(texto, strip=True)


def validar_archivo(imagen):
    """Valida tipo y tamaño de archivo."""
    valid_types = ['png', 'jpeg']
    ext = imghdr.what(imagen)
    if ext not in valid_types:
        raise ValidationError(f"Tipo de archivo no permitido ({ext}). Solo PNG o JPEG.")
    if imagen.size > 2 * 1024 * 1024:
        raise ValidationError("La imagen supera el tamaño permitido (2MB).")
    if os.path.splitext(imagen.name)[1].lower() not in ['.png', '.jpg', '.jpeg']:
        raise ValidationError("Extensión de archivo no permitida.")


# ============================================================
# 🧱 Vista principal: registro de organización y usuario
# ============================================================

@api_view(["POST"])
@parser_classes([MultiPartParser, FormParser])
@permission_classes([AllowAny])
@throttle_classes([RegistroWizardThrottle])
def registro_wizard(request):
    """
    Crea todos los registros iniciales (organización + usuario + info).
    Esta es la única vista pública, protegida con throttling y validación estricta.
    """
    try:
        # ===============================================================
        # 🧠 HONEYPOT ANTI-BOT (campo invisible en frontend)
        # ===============================================================
        if request.data.get("apellidoExtra"):  # si un bot llena este campo
            logger.warning("Intento de bot detectado (honeypot activado).")
            return Response({"error": "Petición sospechosa detectada."}, status=status.HTTP_400_BAD_REQUEST)

        # ===============================================================
        # 🔍 VALIDACIÓN DE ORGANIZACIÓN
        # ===============================================================
        nombre_org = limpiar_texto(request.data.get("nombre"))
        if not nombre_org:
            return Response({"error": "Falta el nombre de la organización."}, status=status.HTTP_400_BAD_REQUEST)

        if len(nombre_org) > 255:
            return Response({"error": "El nombre de la organización es demasiado largo."}, status=status.HTTP_400_BAD_REQUEST)

        if Organizacion.objects.filter(nombre__iexact=nombre_org).exists():
            return Response({"error": "Organización ya registrada."}, status=status.HTTP_400_BAD_REQUEST)

        # ===============================================================
        # ⚙️ TRANSACCIÓN SEGURA
        # ===============================================================
        with transaction.atomic():
            # -----------------------------------------------------------
            # 🖼️ Marca de Agua
            # -----------------------------------------------------------
            marca_agua = None
            if "marcaAgua" in request.FILES:
                archivo = request.FILES["marcaAgua"]
                validar_archivo(archivo)
                marca_agua = ImagenMarcaAgua.objects.create(imagen=archivo)
            else:
                default_img = "marca_agua/white.png"
                if os.path.exists(os.path.join(settings.MEDIA_ROOT, default_img)):
                    marca_agua = ImagenMarcaAgua.objects.create(imagen=default_img)

            # -----------------------------------------------------------
            # 📄 Info Cotización
            # -----------------------------------------------------------
            cotizacion_data = json.loads(request.data.get("cotizacion", "{}"))
            info_cotizacion = InfoCotizacion.objects.create(
                nombreFormato=limpiar_texto(cotizacion_data.get("nombreFormato")),
                version=limpiar_texto(cotizacion_data.get("version")),
                fechaEmision=cotizacion_data.get("fechaEmision"),
                tituloDocumento=limpiar_texto(cotizacion_data.get("tituloDocumento")),
                mensajePropuesta=limpiar_texto(cotizacion_data.get("mensajePropuesta")),
                termino=limpiar_texto(cotizacion_data.get("termino")),
                avisos=limpiar_texto(cotizacion_data.get("avisos")),
                imagenMarcaAgua=marca_agua,
            )

            # -----------------------------------------------------------
            # 🧾 Info Orden de Trabajo
            # -----------------------------------------------------------
            orden_data = json.loads(request.data.get("orden", "{}"))
            info_orden = InfoOrdenTrabajo.objects.create(
                nombreFormato=limpiar_texto(orden_data.get("nombreFormato")),
                version=limpiar_texto(orden_data.get("version")),
                fechaEmision=orden_data.get("fechaEmision"),
                tituloDocumento=limpiar_texto(orden_data.get("tituloDocumento")),
                prefijo=limpiar_texto(orden_data.get("prefijo")),
                imagenMarcaAgua=marca_agua,
            )

            # -----------------------------------------------------------
            # ⚙️ Info Sistema
            # -----------------------------------------------------------
            sistema_data = json.loads(request.data.get("configuracion", "{}"))
            iva = Iva.objects.filter(id=1).first()
            moneda = TipoMoneda.objects.filter(id=1).first()
            info_sistema = InfoSistema.objects.create(
                tipoCambioDolar=sistema_data.get("tipoCambioDolar"),
                tipoMoneda=moneda,
                iva=iva,
            )

            # -----------------------------------------------------------
            # 🏢 Organización
            # -----------------------------------------------------------
            logo = request.FILES.get("logo_organizacion")
            validar_archivo(logo)

            organizacion = Organizacion.objects.create(
                nombre=nombre_org,
                ciudad=limpiar_texto(request.data.get("ciudad")),
                colonia=limpiar_texto(request.data.get("colonia")),
                numero=limpiar_texto(request.data.get("numero")),
                calle=limpiar_texto(request.data.get("calle")),
                codigoPostal=limpiar_texto(request.data.get("codigoPostal")),
                estado=limpiar_texto(request.data.get("estado")),
                slogan=limpiar_texto(request.data.get("slogan")),
                telefono=limpiar_texto(request.data.get("telefono")),
                pagina=limpiar_texto(request.data.get("pagina")),
                RegimenFiscal_id=request.data.get("regimenFiscal"),
                logo=logo,
                infoCotizacion=info_cotizacion,
                infoOrdenTrabajo=info_orden,
                infoSistema=info_sistema,
            )

            # -----------------------------------------------------------
            # 👤 Usuario
            # -----------------------------------------------------------
            username = request.data.get("username")
            password = request.data.get("password")
            email = request.data.get("email")

            if not username or not password:
                raise ValidationError("Usuario o contraseña faltantes.")
            if len(password) < 8:
                raise ValidationError("La contraseña debe tener al menos 8 caracteres.")
            if User.objects.filter(username=username).exists():
                return Response({"error": "Usuario ya existe."}, status=status.HTTP_400_BAD_REQUEST)

            user = User.objects.create_user(
                username=username,
                email=email,
                first_name=limpiar_texto(request.data.get("first_name")),
                last_name=limpiar_texto(request.data.get("last_name")),
                password=password,
                rol_id=3,  # Administradororganizacion
                organizacion=organizacion,
            )

        # ===============================================================
        # ✅ RESPUESTA FINAL
        # ===============================================================
        return Response(
            {"mensaje": "Registro completado correctamente.", "usuario": user.username},
            status=status.HTTP_201_CREATED,
        )

    except ValidationError as e:
        return Response({"error": str(e)}, status=status.HTTP_400_BAD_REQUEST)
    except Exception as e:
        logger.error(f"❌ Error en registro_wizard: {e}")
        return Response({"error": "Error inesperado en el registro."}, status=status.HTTP_400_BAD_REQUEST)
